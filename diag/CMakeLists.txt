cmake_minimum_required(VERSION 3.24)
project(sdk_exit_diag LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
foreach(name load_only init_cleanup)
  add_executable(${name} ${name}.cpp)
  target_include_directories(${name} SYSTEM PRIVATE "${SDK_DIR}/h")
  target_link_libraries(${name} PRIVATE "${SDK_DIR}/lib/sdk.lib")
  target_compile_definitions(${name} PRIVATE UNICODE _UNICODE)
endforeach()
